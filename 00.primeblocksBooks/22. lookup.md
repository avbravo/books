```java
 public Boolean searchAppConfiguration() {
        Boolean result = Boolean.FALSE;
        try {
/**
* Si deseas que consulte todas las paginas coloque page = 0
**/
            Integer page = 1;
            Integer size = 1;
            Bson filter = new Document("applicative.idapplicative", idapplicative.get().longValue()).append("active", Boolean.TRUE);
            Document sort = new Document("applicative.idapplicative", 1);

            List<Appconfiguration> list = appconfigurationClient.lookup(
                    EncodeUtil.encodeBson(filter),
                    EncodeUtil.encodeBson(sort),
                    page, size);

            if (list == null || list.isEmpty()) {
                FacesUtil.warningMessage(rf.fromCore("appconfiguration.restrictivelogin"));

                return Boolean.FALSE;
            } else {
                appconfigurationLogged = list.get(0);
                if (appconfigurationLogged.getRole() == null || appconfigurationLogged.getRole().isEmpty()) {
                    FacesUtil.warningMessage(rf.fromCore("appconfiguration.nohaverole"));
                    return Boolean.FALSE;
                }
                Boolean haveActive = Boolean.FALSE;
                for (Role r : appconfigurationLogged.getRole()) {
                    if (r.getActive()) {
                        haveActive = Boolean.TRUE;
                        break;
                    }
                }
                if (!haveActive) {
                    FacesUtil.warningMessage(rf.fromCore("appconfiguration.nohaveactiverole"));
                    return Boolean.FALSE;
                }

                return haveActive;
            }

        } catch (Exception e) {
            FacesUtil.errorMessage(FacesUtil.nameOfMethod() + " " + e.getLocalizedMessage());

        }
        return result;
    }
```
