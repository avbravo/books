
# Drag and Drop filas en un datatable


Pasos:

1. Agregue  draggableRows="true"

2. Utilice  <p:ajax event="rowReorder"

3. Añada una columna con un icono

```
   <p:column class="column1">
            <i class="pi pi-arrows-v"></i>
    </p:column>
```
4. En el controller añada


```
                                               >

                                    <p:menuitem update=":growl, dataTableTarea" styleClass="ui-button-flat"

                                                icon="pi pi-arrow-up" >
                                    </p:menuitem>


                                    <p:menuitem  styleClass="ui-button-flat"
                                                 update=":growl, dataTableTarea"

                                                 value=""

                                    </p:menuitem>
                                </p:menuButton>
</p:column>
```

En el controller

```java

    // <editor-fold defaultstate="collapsed" desc="void upTarea(Tarjeta tarjeta, Tarea tarea))">
    public void upTarea(Tarea tarea) {
        try {
         

            var index = 0;

            Integer c = 0;
            for (Tarea t : tarjetaTareaSelected.getTarea()) {
                if (t.getTarea().equals(tarea.getTarea())) {
                    index = c;
                }
                c++;
            }

         

            if (index != 0) {
                tarjetaTareaSelected.getTarea().remove(tarea);
                tarjetaTareaSelected.getTarea().add((index - 1), tarea);
                c = 0;
                for (Tarea t : tarjetaTareaSelected.getTarea()) {
                    tarjetaTareaSelected.getTarea().get(c).setOrden(c);
                    c++;
                }
            }
          
            isChangeInRowDatatableTarea = Boolean.TRUE;
            PrimeFaces.current().ajax().update("dataTableTarea");

        } catch (Exception e) {
            FacesUtil.errorMessage(FacesUtil.nameOfClassAndMethod() + " " + e.getLocalizedMessage());
        }
    }
// </editor-fold>
```
