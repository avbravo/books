## SCROOL DIALOGO

MEDIANTE CSS

Agregue a jmoordbcore.css

```css

.jmoordbcore-dialog-responsive{
    
    minHeight:40px;
    width:400px;
    minWidth:100px;
    
}
.jmoordbcore-scrollPanel-responsive{
    
   min-width: 100px;
   min-height: 40px ;
   max-width: 400px; 
   max-height: 750px
    
}

```

Agregue en los dialgos

```
<p:dialog header="#{msg['dialogo.sprint']}" widgetVar="overlayPanelSprint" styleClass="jmoordbcore-dialog-responsive" showEffect="fade" modal="true">

```

y un scrollPanel

```
<p:scrollPanel styleClass="jmoordbcore-scrollPanel-responsive"  mode="native">

```

Agregar al datatable rowSelectedMode = "add"


```
<p:dataTable var="item" value="#{sprintFaces.sprintLazyDataModel}"
     binding="#{sprintFaces.dataTable}"
     id="dataTable"
     widgetVar="widgetVardataTable"
     lazy="true"
     paginator="true"
     rows="#{sprintFaces.rowPageSmall.get()}"
     paginatorTemplate="{RowsPerPageDropdown} {FirstPageLink} {PreviousPageLink} {CurrentPageReport} {NextPageLink} {LastPageLink}"
     paginatorPosition="both"
    reflow="true"
     rowKey="#{item.idsprint}"
     selection="#{sprintFaces.sprintSelected}"
     rowSelectMode="add"

     >

```
