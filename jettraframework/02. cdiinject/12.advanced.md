**3. Annotation Processor**

This is the core of the dependency injection mechanism.  This is a complex class.
```java
// DependencyInjectionProcessor.java
import com.google.auto.service.AutoService;
import javax.annotation.processing.*;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.ElementKind;
import javax.lang.model.element.TypeElement;
import javax.tools.Diagnostic;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

@SupportedAnnotationTypes({"com.example.Inject", "com.example.Service", "com.example.Component"})
@SupportedSourceVersion(SourceVersion.RELEASE_17)
@AutoService(Processor.class)
public class DependencyInjectionProcessor extends AbstractProcessor {

    private final Map<String, Object> instances = new HashMap<>();

    @Override
    public synchronized void init(ProcessingEnvironment processingEnv) {
        super.init(processingEnv);
    }

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        if (annotations.isEmpty()) {
            return false;
        }

        try {
            processServices(roundEnv);
            processComponents(roundEnv);
            processDependencies(roundEnv);
        } catch (Exception e) {
            processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, "Error processing annotations: " + e.getMessage());
            e.printStackTrace(); // Print the stack trace for debugging
            return true; // Claim the annotation to prevent further processing
        }


        return true;
    }

    private void processServices(RoundEnvironment roundEnv) {
        for (Element element : roundEnv.getElementsAnnotatedWith(Service.class)) {
            if (element.getKind() == ElementKind.CLASS) {
                try {
                    createInstance(element.asType().toString());
                    processingEnv.getMessager().printMessage(Diagnostic.Kind.NOTE, "Service " + element.getSimpleName() + " processed.");

                } catch (Exception e) {
                    processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, "Error creating service " + element.getSimpleName() + ": " + e.getMessage());
                    e.printStackTrace();
                }
            }
        }
    }

     private void processComponents(RoundEnvironment roundEnv) {
            for (Element element : roundEnv.getElementsAnnotatedWith(Component.class)) {
                if (element.getKind() == ElementKind.CLASS) {
                    try {
                        createInstance(element.asType().toString());
                        processingEnv.getMessager().printMessage(Diagnostic.Kind.NOTE, "Component " + element.getSimpleName() + " processed.");

                    } catch (Exception e) {
                        processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, "Error creating Component " + element.getSimpleName() + ": " + e.getMessage());
                        e.printStackTrace();
                    }
                }
            }
        }


    private void processDependencies(RoundEnvironment roundEnv) {
        for (Element element : roundEnv.getElementsAnnotatedWith(Inject.class)) {
            if (element.getKind() == ElementKind.FIELD) {
                try {
                    injectDependency(element);
                    processingEnv.getMessager().printMessage(Diagnostic.Kind.NOTE, "Dependency " + element.getSimpleName() + " injected.");

                } catch (Exception e) {
                    processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, "Error injecting dependency " + element.getSimpleName() + ": " + e.getMessage());
                    e.printStackTrace();
                }
            }
        }
    }

    private void injectDependency(Element fieldElement) throws Exception {
        String fieldType = fieldElement.asType().toString();
        Object dependency = getInstance(fieldType);

        if (dependency == null) {
            throw new IllegalStateException("No instance found for dependency: " + fieldType);
        }

        Element enclosingElement = fieldElement.getEnclosingElement();
        String className = enclosingElement.asType().toString();
        Object targetObject = getInstance(className);

        if (targetObject == null) {
            throw new IllegalStateException("No instance found for class: " + className);
        }

        java.lang.reflect.Field field = targetObject.getClass().getDeclaredField(fieldElement.getSimpleName().toString());
        field.setAccessible(true); // Allow access to private fields
        field.set(targetObject, dependency);
        processingEnv.getMessager().printMessage(Diagnostic.Kind.NOTE, "Injected " + fieldType + " into " + className + "." );

    }


    private Object createInstance(String className) throws Exception {
        if (instances.containsKey(className)) {
            return instances.get(className);
        }

        Class<?> clazz = Class.forName(className);
        Object instance = clazz.getDeclaredConstructor().newInstance(); // Use default constructor

        instances.put(className, instance);
        processingEnv.getMessager().printMessage(Diagnostic.Kind.NOTE, "Created instance of " + className);
        return instance;
    }

    public Object getInstance(String className) {
        return instances.get(className);
    }
}
```

Key improvements and explanations:

*   **Error Handling:** Includes comprehensive error handling with try-catch blocks and detailed error messages using `processingEnv.getMessager().printMessage()`.  This is *crucial* for debugging annotation processors.  The stack trace is also printed to make debugging *much* easier.
*   **Annotation Handling:**  Properly handles the `@Service`, `@Component` and `@Inject` annotations.
*   **Instance Management:**  Uses a `HashMap` (`instances`) to store created instances, ensuring singletons (by default).
*   **Dependency Injection:**  The `injectDependency` method finds the target object (the class containing the `@Inject` field) and the dependency to inject, then uses reflection to set the field value.  It also includes important checks to ensure instances exist.
*   **Reflection:** Uses reflection (`java.lang.reflect.Field`) to access and set the value of the injected fields.  This allows injection into private fields, but requires `field.setAccessible(true);`.
*   **`AutoService`:**  The `@AutoService(Processor.class)` annotation automatically registers the processor with the Java compiler.  This is much cleaner than manually configuring the processor.
*   **Error messages:** The logging uses the processingEnv's messenger to correctly create error messages that are visible at compile time.
*   **Singleton by default**: It creates singletons. If another instance of the service has to be created the HashMap that contains the services should allow the existence of more than one service

**Important Considerations for the Annotation Processor:**

*   **Immutability:** Be *very* careful about modifying elements (classes, fields, etc.) directly within the processor.  This can lead to unexpected compiler behavior. The processor is mainly for *reading* information.  If you need to modify the source code, you should use a code generation library like Javapoet.  This example focuses on *runtime* dependency injection rather than generating new code.
*   **Incremental Compilation:**  Annotation processors can significantly impact build times.  Consider how your processor interacts with incremental compilation.  You might need to declare specific dependencies or use APIs to support incremental processing effectively.
*   **Testing:**  Testing annotation processors is challenging.  You often need to create dummy source files and compile them to test the processor's behavior.  Libraries like Truth and Compile Testing can help.
*   **Complexity:**  Annotation processors are powerful but complex.  Carefully plan your design, handle errors gracefully, and write thorough tests.

// PersonRepository.java
import com.example.Service;
import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import org.bson.Document;
import org.bson.types.ObjectId;

import static com.mongodb.client.model.Filters.eq;

@Service
public class PersonRepository {

    private final MongoClient mongoClient;
    private final MongoDatabase database;
    private final MongoCollection<Document> collection;

    public PersonRepository() {
        // Replace with your MongoDB connection string and database name
        String connectionString = "mongodb://localhost:27017";
        String databaseName = "mydatabase";

        this.mongoClient = MongoClients.create(connectionString);
        this.database = mongoClient.getDatabase(databaseName);
        this.collection = database.getCollection("persons");
    }

    public Person create(Person person) {
        Document doc = new Document("name", person.getName())
                .append("age", person.getAge());
        collection.insertOne(doc);
        person.setId(doc.getObjectId("_id").toString());
        return person;
    }

    public Person get(String id) {
        Document doc = collection.find(eq("_id", new ObjectId(id))).first();
        if (doc == null) {
            return null;
        }
        Person person = new Person();
        person.setId(doc.getObjectId("_id").toString());
        person.setName(doc.getString("name"));
        person.setAge(doc.getInteger("age"));
        return person;
    }

    public Person update(String id, Person person) {
        Document updateDoc = new Document("name", person.getName())
                .append("age", person.getAge());
        collection.updateOne(eq("_id", new ObjectId(id)), new Document("$set", updateDoc));
        person.setId(id);
        return person;
    }

    public void delete(String id) {
        collection.deleteOne(eq("_id", new ObjectId(id)));
    }
}

import com.example.Inject;
import com.example.Component;
import com.sun.net.httpserver.HttpExchange;
import com.sun.net.httpserver.HttpHandler;

import java.io.IOException;
import java.io.OutputStream;
import java.nio.charset.StandardCharsets;
import com.google.gson.Gson;


@Component
public class PersonHandler implements HttpHandler {

    @Inject
    private PersonService personService;

    @Override
    public void handle(HttpExchange exchange) throws IOException {
        String path = exchange.getRequestURI().getPath();
        String method = exchange.getRequestMethod();
        String response = "";

        if ("/person".equals(path) && "POST".equals(method)) {
            // Create Person
            String requestBody = new String(exchange.getRequestBody().readAllBytes(), StandardCharsets.UTF_8);
            Gson gson = new Gson();
            Person person = gson.fromJson(requestBody, Person.class);
            Person createdPerson = personService.createPerson(person);
            response = gson.toJson(createdPerson);
        } else if (path.startsWith("/person/") && "GET".equals(method)) {
            // Get Person by ID
            String id = path.substring("/person/".length());
            Person person = personService.getPerson(id);
            if (person != null) {
                Gson gson = new Gson();
                response = gson.toJson(person);
            } else {
                response = "Person not found";
                exchange.sendResponseHeaders(404, 0);
                OutputStream os = exchange.getResponseBody();
                os.write(response.getBytes());
                os.close();
                return;

            }

        }  else if (path.startsWith("/person/") && "PUT".equals(method)) {
             String id = path.substring("/person/".length());
            String requestBody = new String(exchange.getRequestBody().readAllBytes(), StandardCharsets.UTF_8);
            Gson gson = new Gson();
            Person person = gson.fromJson(requestBody, Person.class);
            Person updatedPerson = personService.updatePerson(id, person);
            response = gson.toJson(updatedPerson);
        } else if (path.startsWith("/person/") && "DELETE".equals(method)) {
            String id = path.substring("/person/".length());
            personService.deletePerson(id);
            response = "Person deleted";

        } else {
            response = "Invalid request";
            exchange.sendResponseHeaders(400, 0);
             OutputStream os = exchange.getResponseBody();
                os.write(response.getBytes());
                os.close();
                return;
        }

        exchange.sendResponseHeaders(200, response.getBytes().length);
        OutputStream os = exchange.getResponseBody();
        os.write(response.getBytes());
        os.close();
    }
}

import com.example.Component;
import com.sun.net.httpserver.HttpServer;
import java.io.IOException;
import java.net.InetSocketAddress;

@Component
public class Main {

    public static void main(String[] args) throws IOException {
        try {
            // This will trigger the annotation processor to run and inject dependencies.
            Class.forName("com.example.PersonHandler");

            System.err.println("Error: Could not load PersonHandler.  Ensure annotation processing is configured correctly.");
            e.printStackTrace();
            return;

        HttpServer server = HttpServer.create(new InetSocketAddress(8080), 0);
        // Get the instance of PersonHandler created by the annotation processor
        DependencyInjectionProcessor processor = new DependencyInjectionProcessor();
        PersonHandler personHandler = (PersonHandler) processor.getInstance("com.example.PersonHandler");

        if (personHandler == null) {
            System.err.println("Error: PersonHandler not found in dependency injection container.  Check your annotation processing configuration.");
            return;
        }


        server.createContext("/person", personHandler); // Use the injected instance
        server.setExecutor(null); // creates a default executor
