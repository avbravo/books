### 1. Clase `Persona`

Primero, definimos la clase `Persona` con algunos atributos básicos:

```java
package com.example.model;

public class Persona {
    private int id;
    private String nombre;
    private int edad;

    public Persona(int id, String nombre, int edad) {
        this.id = id;
        this.nombre = nombre;
        this.edad = edad;
    }

    public int getId() {
        return id;
    }

    public void setId(int id) {
        this.id = id;
    }

    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public int getEdad() {
        return edad;
    }

    public void setEdad(int edad) {
        this.edad = edad;
    }

    @Override
    public String toString() {
        return "Persona{id=" + id + ", nombre='" + nombre + "', edad=" + edad + "}";
    }
}
```

### 2. Clase con Endpoints CRUD (`PersonaEndpoints`)

Definimos los métodos CRUD anotados con `@Endpoint`. Estos métodos manejarán las operaciones sobre una lista de personas.

```java
package com.example.endpoints;

import com.example.annotations.Endpoint;
import com.example.model.Persona;

import java.util.ArrayList;
import java.util.List;

public class PersonaEndpoints {

    private List<Persona> personas = new ArrayList<>();
    private int nextId = 1;

    // Crear una nueva persona (POST)
    @Endpoint(path = "/personas", method = "POST")
    public String crearPersona(String body) {
        String[] datos = body.split(",");
        if (datos.length != 2) {
            return "Error: El cuerpo debe tener el formato 'nombre,edad'";
        }
        String nombre = datos[0];
        int edad = Integer.parseInt(datos[1]);
        Persona persona = new Persona(nextId++, nombre, edad);
        personas.add(persona);
        return "Creado: " + persona.toString();
    }

    // Obtener todas las personas (GET)
    @Endpoint(path = "/personas", method = "GET")
    public String obtenerPersonas() {
        return personas.toString();
    }

    // Obtener una persona por ID (GET)
    @Endpoint(path = "/personas/{id}", method = "GET")
    public String obtenerPersonaPorId(int id) {
        for (Persona persona : personas) {
            if (persona.getId() == id) {
                return persona.toString();
            }
        }
        return "Persona no encontrada";
    }

    // Actualizar una persona (PUT)
    @Endpoint(path = "/personas/{id}", method = "PUT")
    public String actualizarPersona(int id, String body) {
        String[] datos = body.split(",");
        if (datos.length != 2) {
            return "Error: El cuerpo debe tener el formato 'nombre,edad'";
        }
        String nombre = datos[0];
        int edad = Integer.parseInt(datos[1]);
        for (Persona persona : personas) {
            if (persona.getId() == id) {
                persona.setNombre(nombre);
                persona.setEdad(edad);
                return "Actualizado: " + persona.toString();
            }
        }
        return "Persona no encontrada";
    }

    // Eliminar una persona (DELETE)
    @Endpoint(path = "/personas/{id}", method = "DELETE")
    public String eliminarPersona(int id) {
        for (Persona persona : personas) {
            if (persona.getId() == id) {
                personas.remove(persona);
                return "Eliminado: Persona con ID " + id;
            }
        }
        return "Persona no encontrada";
    }
}
```

### 3. Modificaciones en el Procesador de Anotaciones

El procesador de anotaciones necesita adaptarse para manejar parámetros en las rutas (como `{id}`) y cuerpos de solicitud (`body`). Aquí está la versión actualizada del procesador:

```java
package com.example.processor;

import com.example.annotations.Endpoint;
import com.sun.net.httpserver.HttpHandler;

import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.ExecutableElement;
import javax.lang.model.element.TypeElement;
import javax.tools.JavaFileObject;
import java.io.IOException;
import java.io.PrintWriter;
import java.util.Set;

@SupportedAnnotationTypes("com.example.annotations.Endpoint")
@SupportedSourceVersion(SourceVersion.RELEASE_8)
public class EndpointProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        StringBuilder generatedCode = new StringBuilder();
        generatedCode.append("package com.example.generated;\n\n");
        generatedCode.append("import com.sun.net.httpserver.HttpServer;\n");
        generatedCode.append("import com.sun.net.httpserver.HttpHandler;\n");
        generatedCode.append("import java.io.OutputStream;\n");
        generatedCode.append("import java.net.InetSocketAddress;\n");
        generatedCode.append("import java.io.IOException;\n");
        generatedCode.append("import java.nio.charset.StandardCharsets;\n\n");

        generatedCode.append("public class EndpointRegistry {\n");
        generatedCode.append("    public static void registerEndpoints(HttpServer server) {\n");

        for (Element element : roundEnv.getElementsAnnotatedWith(Endpoint.class)) {
            if (element instanceof ExecutableElement) {
                ExecutableElement method = (ExecutableElement) element;
                Endpoint endpoint = method.getAnnotation(Endpoint.class);

                String path = endpoint.path();
                String httpMethod = endpoint.method();
                String methodName = method.getSimpleName().toString();
                String className = ((TypeElement) method.getEnclosingElement()).getQualifiedName().toString();

                generatedCode.append("        server.createContext(\"").append(path).append("\", new HttpHandler() {\n");
                generatedCode.append("            @Override\n");
                generatedCode.append("            public void handle(com.sun.net.httpserver.HttpExchange exchange) throws IOException {\n");
                generatedCode.append("                if (!exchange.getRequestMethod().equalsIgnoreCase(\"").append(httpMethod).append("\")) {\n");
                generatedCode.append("                    exchange.sendResponseHeaders(405, -1);\n");
                generatedCode.append("                    return;\n");
                generatedCode.append("                }\n");

                // Manejo de parámetros en la ruta
                if (path.contains("{")) {
                    generatedCode.append("                String[] pathParts = exchange.getRequestURI().getPath().split(\"/\");\n");
                    generatedCode.append("                int id = Integer.parseInt(pathParts[pathParts.length - 1]);\n");
                }

                // Manejo del cuerpo de la solicitud
                if (httpMethod.equals("POST") || httpMethod.equals("PUT")) {
                    generatedCode.append("                String body = new String(exchange.getRequestBody().readAllBytes(), StandardCharsets.UTF_8);\n");
                }

                generatedCode.append("                try {\n");
                generatedCode.append("                    ").append(className).append(" instance = new ").append(className).append("();\n");

                // Llamada al método con parámetros
                if (path.contains("{")) {
                    if (httpMethod.equals("POST") || httpMethod.equals("PUT")) {
                        generatedCode.append("                    String response = instance.").append(methodName).append("(id, body);\n");
                    } else {
                        generatedCode.append("                    String response = instance.").append(methodName).append("(id);\n");
                    }
                } else if (httpMethod.equals("POST") || httpMethod.equals("PUT")) {
                    generatedCode.append("                    String response = instance.").append(methodName).append("(body);\n");
                } else {
                    generatedCode.append("                    String response = instance.").append(methodName).append("();\n");
                }

                generatedCode.append("                    exchange.sendResponseHeaders(200, response.getBytes().length);\n");
                generatedCode.append("                    OutputStream os = exchange.getResponseBody();\n");
                generatedCode.append("                    os.write(response.getBytes());\n");
                generatedCode.append("                    os.close();\n");
                generatedCode.append("                } catch (Exception e) {\n");
                generatedCode.append("                    exchange.sendResponseHeaders(500, -1);\n");
                generatedCode.append("                }\n");
                generatedCode.append("            }\n");
                generatedCode.append("        });\n");
            }
        }

        generatedCode.append("    }\n");
        generatedCode.append("}\n");

        try {
            JavaFileObject file = processingEnv.getFiler().createSourceFile("com.example.generated.EndpointRegistry");
            try (PrintWriter out = new PrintWriter(file.openWriter())) {
                out.print(generatedCode.toString());
            }
        } catch (IOException e) {
            e.printStackTrace();
        }

        return true;
    }
}
```
