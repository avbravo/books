Para añadir endpoints CRUD usando `HttpHandler` en Java, podemos implementar un servidor HTTP básico utilizando la clase `com.sun.net.httpserver.HttpServer`. Este servidor manejará solicitudes HTTP y ejecutará operaciones CRUD sobre las entidades `Persona` y `Pais`. A continuación, te muestro cómo integrar esto en el programa existente.

---

### **Estructura del programa**

1. **Servidor HTTP**:
   - Usaremos `HttpServer` para crear un servidor que escuche en un puerto específico.
   - Definiremos rutas (endpoints) para manejar operaciones CRUD para `Persona` y `Pais`.

2. **Endpoints**:
   - `/personas`: Maneja operaciones CRUD para la entidad `Persona`.
   - `/paises`: Maneja operaciones CRUD para la entidad `Pais`.

3. **Repositorios**:
   - Los repositorios (`PersonaRepository` y `PaisRepository`) seguirán siendo responsables de interactuar con MongoDB.

4. **Clase principal (`Main`)**:
   - Inicia el servidor HTTP y registra los endpoints.

import com.mongodb.client.MongoClient;

// Entidad Persona
class Persona {
    private String id;
    private String nombre;
    private int edad;

    public Persona(String nombre, int edad) {
        this.nombre = nombre;
        this.edad = edad;
    }

    public Persona(String id, String nombre, int edad) {
        this.id = id;
        this.nombre = nombre;
        this.edad = edad;
    }

    public String getId() {
        return id;
    }

    public String getNombre() {
        return nombre;
    }

    public int getEdad() {
        return edad;
    }

    @Override
    public String toString() {
        return "Persona{id='" + id + "', nombre='" + nombre + "', edad=" + edad + "}";
    }
}

// Interfaz común para la conexión a MongoDB
interface MongoConnection {
    MongoDatabase getDatabase(String databaseName);
}

// Implementación concreta de la conexión
class MongoConnectionImpl implements MongoConnection {

    private final MongoClient mongoClient;

    public MongoConnectionImpl(String connectionString) {
        this.mongoClient = MongoClients.create(connectionString);
    }

    @Override
    public MongoDatabase getDatabase(String databaseName) {
        return mongoClient.getDatabase(databaseName);
    }
}

// Repositorio para la entidad Persona
class PersonaRepository {

    private final MongoCollection<Document> collection;

    public PersonaRepository(MongoConnection mongoConnection) {
        MongoDatabase database = mongoConnection.getDatabase("test");
        this.collection = database.getCollection("personas");
    }

    public String crearPersona(Persona persona) {
        Document doc = new Document("nombre", persona.getNombre())
                .append("edad", persona.getEdad());
        InsertOneResult result = collection.insertOne(doc);
        return result.getInsertedId().asObjectId().getValue().toString();
    }

    public Persona obtenerPersonaPorId(String id) {
        Document doc = collection.find(new Document("_id", new ObjectId(id))).first();
        if (doc == null) return null;
        return new Persona(
                doc.getObjectId("_id").toString(),
                doc.getString("nombre"),
                doc.getInteger("edad")
        );
    }

    public boolean actualizarPersona(String id, Persona persona) {
        UpdateResult result = collection.updateOne(
                new Document("_id", new ObjectId(id)),
                new Document("$set", new Document("nombre", persona.getNombre()).append("edad", persona.getEdad()))
        );
        return result.getModifiedCount() > 0;
    }

    public boolean eliminarPersona(String id) {
        DeleteResult result = collection.deleteOne(new Document("_id", new ObjectId(id)));
        return result.getDeletedCount() > 0;
    }
}

// Sistema de inyección de dependencias personalizado
class DependencyInjector {

    private final Map<Class<?>, Supplier<?>> producers = new HashMap<>();

    public <T> void registerProducer(Class<T> type, Supplier<T> producer) {
        producers.put(type, producer);
    }

    @SuppressWarnings("unchecked")
    public <T> T getInstance(Class<T> type) {
        Supplier<?> producer = producers.get(type);
        if (producer == null) {
            throw new IllegalArgumentException("No producer registered for type: " + type.getName());
        }
        return (T) producer.get();
    }
}

// Clase principal para probar el sistema
public class Main {

    public static void main(String[] args) throws IOException {
        // Crear el inyector de dependencias
        DependencyInjector injector = new DependencyInjector();

        // Registrar un productor para MongoConnection
        String connectionString = "mongodb://localhost:27017"; // Cambia esto según tu configuración
        injector.registerProducer(MongoConnection.class, () -> new MongoConnectionImpl(connectionString));

        // Obtener instancias de los repositorios
        MongoConnection mongoConnection = injector.getInstance(MongoConnection.class);
        PersonaRepository personaRepo = new PersonaRepository(mongoConnection);
        PaisRepository paisRepo = new PaisRepository(mongoConnection);

        // Crear el servidor HTTP
        HttpServer server = HttpServer.create(new InetSocketAddress(8080), 0);

        // Endpoint para Persona
        server.createContext("/personas", new PersonaHandler(personaRepo));
        // Endpoint para Pais
        server.createContext("/paises", new PaisHandler(paisRepo));

        // Iniciar el servidor
        server.setExecutor(null); // Usa el executor por defecto
        server.start();
        System.out.println("Servidor HTTP iniciado en http://localhost:8080");
    }
}

// Handler para la entidad Persona
class PersonaHandler implements HttpHandler {

    private final PersonaRepository personaRepo;

    public PersonaHandler(PersonaRepository personaRepo) {
        this.personaRepo = personaRepo;
    }

    @Override
    public void handle(HttpExchange exchange) throws IOException {
        String method = exchange.getRequestMethod();
        String response = "";

        try {
            switch (method) {
                case "POST": // Crear una nueva persona
                    String body = new String(exchange.getRequestBody().readAllBytes());
                    String[] parts = body.split(",");
                    String nombre = parts[0];
                    int edad = Integer.parseInt(parts[1]);
                    String id = personaRepo.crearPersona(new Persona(nombre, edad));
                    response = "Persona creada con ID: " + id;
                    exchange.sendResponseHeaders(201, response.length());
                    break;

                case "GET": // Obtener una persona por ID
                    String query = exchange.getRequestURI().getQuery();
                    String idParam = query.split("=")[1];
                    Persona persona = personaRepo.obtenerPersonaPorId(idParam);
                    response = persona != null ? persona.toString() : "Persona no encontrada";
                    exchange.sendResponseHeaders(200, response.length());
                    break;

                case "PUT": // Actualizar una persona
                    String putBody = new String(exchange.getRequestBody().readAllBytes());
                    String[] putParts = putBody.split(",");
                    String updateId = putParts[0];
                    String updateNombre = putParts[1];
                    int updateEdad = Integer.parseInt(putParts[2]);
                    boolean updated = personaRepo.actualizarPersona(updateId, new Persona(updateNombre, updateEdad));
                    response = updated ? "Persona actualizada" : "Persona no encontrada";
                    exchange.sendResponseHeaders(200, response.length());
                    break;

                case "DELETE": // Eliminar una persona
                    String deleteQuery = exchange.getRequestURI().getQuery();
                    String deleteId = deleteQuery.split("=")[1];
                    boolean deleted = personaRepo.eliminarPersona(deleteId);
                    response = deleted ? "Persona eliminada" : "Persona no encontrada";
                    exchange.sendResponseHeaders(200, response.length());
                    break;

                default:
                    response = "Método no soportado";
                    exchange.sendResponseHeaders(405, response.length());
                    break;
            }
        } catch (Exception e) {
            response = "Error: " + e.getMessage();
            exchange.sendResponseHeaders(500, response.length());
        }

        OutputStream os = exchange.getResponseBody();
        os.write(response.getBytes());
        os.close();
    }
}
