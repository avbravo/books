Además, implementaremos repositorios específicos para cada entidad (`PersonaRepository` y `PaisRepository`) que utilicen el sistema de inyección de dependencias (`DependencyInjector`) para obtener la conexión a la base de datos.

### **Estructura del programa**

1. **Entidades**:
   - `Persona`: Representa una persona con atributos como `id`, `nombre` y `edad`.
   - `Pais`: Representa un país con atributos como `id` y `nombre`.

2. **Repositorios**:
   - `PersonaRepository`: Contiene métodos CRUD para la colección `personas`.
   - `PaisRepository`: Contiene métodos CRUD para la colección `paises`.

3. **Inyección de dependencias**:
   - El `DependencyInjector` proporcionará la conexión a MongoDB a los repositorios.

4. **Clase principal (`Main`)**:
   - Ejecuta operaciones CRUD sobre las entidades usando los repositorios.

### **Código completo**

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.result.DeleteResult;
import com.mongodb.client.result.InsertOneResult;
import com.mongodb.client.result.UpdateResult;
import org.bson.Document;
import org.bson.types.ObjectId;

import java.util.HashMap;
import java.util.Map;
import java.util.function.Supplier;

// Entidad Persona
class Persona {
    private String id;
    private String nombre;
    private int edad;

    public Persona(String nombre, int edad) {
        this.nombre = nombre;
        this.edad = edad;
    }

    public Persona(String id, String nombre, int edad) {
        this.id = id;
        this.nombre = nombre;
        this.edad = edad;
    }

    public String getId() {
        return id;
    }

    public String getNombre() {
        return nombre;
    }

    public int getEdad() {
        return edad;
    }

    @Override
    public String toString() {
        return "Persona{id='" + id + "', nombre='" + nombre + "', edad=" + edad + "}";
    }
}

// Entidad Pais
class Pais {
    private String id;
    private String nombre;

    public Pais(String nombre) {
        this.nombre = nombre;
    }

    public Pais(String id, String nombre) {
        this.id = id;
        this.nombre = nombre;
    }

    public String getId() {
        return id;
    }

    public String getNombre() {
        return nombre;
    }

    @Override
    public String toString() {
        return "Pais{id='" + id + "', nombre='" + nombre + "'}";
    }
}

// Interfaz común para la conexión a MongoDB
interface MongoConnection {
    MongoDatabase getDatabase(String databaseName);
}

// Implementación concreta de la conexión
class MongoConnectionImpl implements MongoConnection {

    private final MongoClient mongoClient;

    public MongoConnectionImpl(String connectionString) {
        this.mongoClient = MongoClients.create(connectionString);
    }

    @Override
    public MongoDatabase getDatabase(String databaseName) {
        return mongoClient.getDatabase(databaseName);
    }
}

// Repositorio para la entidad Persona
class PersonaRepository {

    private final MongoCollection<Document> collection;

    public PersonaRepository(MongoConnection mongoConnection) {
        MongoDatabase database = mongoConnection.getDatabase("test");
        this.collection = database.getCollection("personas");
    }

    public String crearPersona(Persona persona) {
        Document doc = new Document("nombre", persona.getNombre())
                .append("edad", persona.getEdad());
        InsertOneResult result = collection.insertOne(doc);
        return result.getInsertedId().asObjectId().getValue().toString();
    }

    public Persona obtenerPersonaPorId(String id) {
        Document doc = collection.find(new Document("_id", new ObjectId(id))).first();
        if (doc == null) return null;
        return new Persona(
                doc.getObjectId("_id").toString(),
                doc.getString("nombre"),
                doc.getInteger("edad")
        );
    }

    public boolean actualizarPersona(String id, Persona persona) {
        UpdateResult result = collection.updateOne(
                new Document("_id", new ObjectId(id)),
                new Document("$set", new Document("nombre", persona.getNombre()).append("edad", persona.getEdad()))
        );
        return result.getModifiedCount() > 0;
    }

    public boolean eliminarPersona(String id) {
        DeleteResult result = collection.deleteOne(new Document("_id", new ObjectId(id)));
        return result.getDeletedCount() > 0;
    }
}

// Sistema de inyección de dependencias personalizado
class DependencyInjector {

    private final Map<Class<?>, Supplier<?>> producers = new HashMap<>();

    public <T> void registerProducer(Class<T> type, Supplier<T> producer) {
        producers.put(type, producer);
    }

    @SuppressWarnings("unchecked")
    public <T> T getInstance(Class<T> type) {
        Supplier<?> producer = producers.get(type);
        if (producer == null) {
            throw new IllegalArgumentException("No producer registered for type: " + type.getName());
        }
        return (T) producer.get();
    }
}

// Clase principal para probar el sistema
public class Main {

    public static void main(String[] args) {
        // Crear el inyector de dependencias
        DependencyInjector injector = new DependencyInjector();

        // Registrar un productor para MongoConnection
        String connectionString = "mongodb://localhost:27017"; // Cambia esto según tu configuración
        injector.registerProducer(MongoConnection.class, () -> new MongoConnectionImpl(connectionString));

        // Obtener instancias de los repositorios
        MongoConnection mongoConnection = injector.getInstance(MongoConnection.class);
        PersonaRepository personaRepo = new PersonaRepository(mongoConnection);
        PaisRepository paisRepo = new PaisRepository(mongoConnection);

        // Operaciones CRUD para Persona
        System.out.println("=== Operaciones CRUD para Persona ===");
        String personaId = personaRepo.crearPersona(new Persona("Juan", 30));
        System.out.println("Persona creada con ID: " + personaId);

        Persona persona = personaRepo.obtenerPersonaPorId(personaId);
        System.out.println("Persona obtenida: " + persona);

        personaRepo.actualizarPersona(personaId, new Persona("Juan Pérez", 31));
        System.out.println("Persona actualizada");

        persona = personaRepo.obtenerPersonaPorId(personaId);
        System.out.println("Persona después de actualizar: " + persona);

        boolean eliminada = personaRepo.eliminarPersona(personaId);
        System.out.println("Persona eliminada: " + eliminada);

        // Operaciones CRUD para Pais
        System.out.println("\n=== Operaciones CRUD para Pais ===");
        String paisId = paisRepo.crearPais(new Pais("México"));
        System.out.println("País creado con ID: " + paisId);

        Pais pais = paisRepo.obtenerPaisPorId(paisId);
        System.out.println("País obtenido: " + pais);

        paisRepo.actualizarPais(paisId, new Pais("Estados Unidos"));
        System.out.println("País actualizado");

        pais = paisRepo.obtenerPaisPorId(paisId);
        System.out.println("País después de actualizar: " + pais);

        boolean eliminado = paisRepo.eliminarPais(paisId);
        System.out.println("País eliminado: " + eliminado);
    }
}
