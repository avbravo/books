```java
import java.io.IOException;
import java.util.Set;
import java.util.stream.Collectors;

import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.Processor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.TypeElement;

import com.google.auto.service.AutoService;
import java.io.PrintWriter;
import java.util.List;
import java.util.Map;
import javax.lang.model.type.ExecutableType;
import javax.tools.Diagnostic;
import javax.tools.JavaFileObject;

@AutoService(Processor.class)
@SupportedAnnotationTypes("com.avbravo.jettraprocessor.annotation.BuilderProperty")
@SupportedSourceVersion(SourceVersion.RELEASE_21)
public class BuilderProcessor extends AbstractProcessor{

   @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (TypeElement annotation : annotations) {

            Set<? extends Element> annotatedElements = roundEnv.getElementsAnnotatedWith(annotation);

            Map<Boolean, List<Element>> annotatedMethods = annotatedElements.stream().collect(Collectors.partitioningBy(element -> ((ExecutableType) element.asType()).getParameterTypes().size() == 1 && element.getSimpleName().toString().startsWith("set")));

            List<Element> setters = annotatedMethods.get(true);
            List<Element> otherMethods = annotatedMethods.get(false);

            otherMethods.forEach(element -> processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, "@BuilderProperty must be applied to a setXxx method with a single argument", element));

            if (setters.isEmpty()) {
                continue;
            }

            String className = ((TypeElement) setters.get(0).getEnclosingElement()).getQualifiedName().toString();

            Map<String, String> setterMap = setters.stream().collect(Collectors.toMap(setter -> setter.getSimpleName().toString(), setter -> ((ExecutableType) setter.asType()).getParameterTypes().get(0).toString()));

            try {
                writeBuilderFile(className, setterMap);
            } catch (IOException e) {
                e.printStackTrace();
            }

        }

        return true;
    }

    private void writeBuilderFile(String className, Map<String, String> setterMap) throws IOException {

        String packageName = null;
        int lastDot = className.lastIndexOf('.');
        if (lastDot > 0) {
            packageName = className.substring(0, lastDot);
        }

        String simpleClassName = className.substring(lastDot + 1);
        String builderClassName = className + "Builder";
        String builderSimpleClassName = builderClassName.substring(lastDot + 1);

        JavaFileObject builderFile = processingEnv.getFiler().createSourceFile(builderClassName);
        try (PrintWriter out = new PrintWriter(builderFile.openWriter())) {

            if (packageName != null) {
                out.print("package ");
                out.print(packageName);
                out.println(";");
                out.println();
            }

            out.print("public class ");
            out.print(builderSimpleClassName);
            out.println(" {");
            out.println();

            out.print("    private ");
            out.print(simpleClassName);
            out.print(" object = new ");
            out.print(simpleClassName);
            out.println("();");
            out.println();

            out.print("    public ");
            out.print(simpleClassName);
            out.println(" build() {");
            out.println("        return object;");
            out.println("    }");
            out.println();

            setterMap.entrySet().forEach(setter -> {
                String methodName = setter.getKey();
                String argumentType = setter.getValue();

                out.print("    public ");
                out.print(builderSimpleClassName);
                out.print(" ");
                out.print(methodName);

                out.print("(");

                out.print(argumentType);
                out.println(" value) {");
                out.print("        object.");
                out.print(methodName);
                out.println("(value);");
                out.println("        return this;");
                out.println("    }");
                out.println();
            });
            out.println("}");
        }
    }
}
```

```java

import com.avbravo.jettraprocessor.annotation.Builder;
import java.io.IOException;
import java.util.Set;

import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.Processor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.TypeElement;

import com.google.auto.service.AutoService;
import java.io.BufferedWriter;
import javax.lang.model.element.Modifier;
import javax.lang.model.element.PackageElement;
import javax.tools.Diagnostic;
import javax.tools.JavaFileObject;

@AutoService(Processor.class)
@SupportedAnnotationTypes("com.avbravo.jettraprocessor.annotation.Builder")
@SupportedSourceVersion(SourceVersion.RELEASE_21)
public class BuilderProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {

        for (Element element : roundEnv.getElementsAnnotatedWith(Builder.class)) {
            JavaFileObject builderClass = null;
            PackageElement packageElement = (PackageElement) element.getEnclosingElement();
            BufferedWriter bufferedWriter = null;
            try {
                String builderName = element.getSimpleName().toString() + "Builder";

                String builderGenName = packageElement.getQualifiedName().toString() + "." + builderName;

//                builderClass = processingEnv.getFiler().createSourceFile(builderName);
                builderClass = processingEnv.getFiler().createSourceFile(builderGenName);
                bufferedWriter = new BufferedWriter(builderClass.openWriter());
                bufferedWriter.append("package ");
                bufferedWriter.append(packageElement.getQualifiedName().toString());
                bufferedWriter.append(";");
                bufferedWriter.newLine();
                bufferedWriter.append("\npublic class ");
                bufferedWriter.append(builderName);
                bufferedWriter.append("{");
                bufferedWriter.newLine();

                bufferedWriter.newLine();
                bufferedWriter.newLine();
                bufferedWriter.append("\n\tprivate ");
                bufferedWriter.append(element.getSimpleName().toString());
                bufferedWriter.append(" object ");
                bufferedWriter.append(" = new " + element.getSimpleName().toString() + "();");

                bufferedWriter.newLine();

                if (element.getKind().isClass()) {
                    for (Element enclosed : element.getEnclosedElements()) {
                        if (enclosed.getKind().isField() & (enclosed.getModifiers().contains(Modifier.PUBLIC)
                                | enclosed.getModifiers().contains(Modifier.PROTECTED))) {
                            bufferedWriter.newLine();
                            bufferedWriter.newLine();
                            bufferedWriter.append("\tpublic ");
                            bufferedWriter.append(builderName);
                            bufferedWriter.append(" set");
                            String field = enclosed.getSimpleName().toString();
                            String s1 = field.substring(0, 1).toUpperCase();
                            String nameCapitalized = s1 + field.substring(1);
                            bufferedWriter.append(nameCapitalized);
                            bufferedWriter.append("(");
                            bufferedWriter.append(enclosed.asType().toString());
                            bufferedWriter.append(" param){");
                            bufferedWriter.newLine();
                            bufferedWriter.append("\tthis.object.");
                            bufferedWriter.append(enclosed.getSimpleName().toString());
                            bufferedWriter.append(" = ");
                            bufferedWriter.append("param;");
                            bufferedWriter.newLine();
                            bufferedWriter.append("\treturn this;\n\t}");
                        }
                    }
                }
                bufferedWriter.newLine();
                bufferedWriter.append("\tpublic ");
                bufferedWriter.append(element.getSimpleName());
                bufferedWriter.append(" build(){");
                bufferedWriter.newLine();
                bufferedWriter.append("\treturn this.object;\n\t}");
                bufferedWriter.newLine();
                bufferedWriter.newLine();
                bufferedWriter.append("}");
                bufferedWriter.close();
            } catch (IOException e) {
                processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, e.toString());
            }

        }
        return false;
    }
}
```

Un proyecto común que utiliza **Java Annotation Processing** para generar código en la misma clase es el uso de bibliotecas como **Lombok**. Lombok es una herramienta muy popular que simplifica la escritura de clases Java al generar automáticamente código repetitivo (como getters, setters, constructores, etc.) mediante el uso de anotaciones.

Aunque Lombok no genera código directamente en la misma clase durante la compilación (en realidad modifica el AST - Abstract Syntax Tree - del compilador), su efecto final es similar: permite escribir menos código y obtener funcionalidades adicionales sin tener que implementarlas manualmente.

Sin embargo, si lo que buscas es un ejemplo más personalizado donde tú mismo implementes un procesador de anotaciones que genere código dentro de la misma clase, aquí tienes un ejemplo básico:

---

### Ejemplo: Procesador de Anotaciones Personalizado

#### 1. Definición de la Anotación
Primero, define una anotación personalizada que se usará para marcar métodos o clases.

```java

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Retention(RetentionPolicy.SOURCE) // Solo disponible en tiempo de compilación
@Target(ElementType.TYPE) // Aplicable a clases
public @interface GenerateToString {
}

```

#### 2. Procesador de Anotaciones
Crea un procesador de anotaciones que genere código automáticamente cuando encuentre la anotación `@GenerateToString`.

```java

import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.TypeElement;
import javax.tools.JavaFileObject;
import java.io.IOException;
import java.io.PrintWriter;
import java.util.Set;

@SupportedAnnotationTypes("GenerateToString")
@SupportedSourceVersion(SourceVersion.RELEASE_8)
public class ToStringProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (Element element : roundEnv.getElementsAnnotatedWith(GenerateToString.class)) {
            if (element instanceof TypeElement) {
                TypeElement typeElement = (TypeElement) element;
                String className = typeElement.getSimpleName().toString();
                String packageName = processingEnv.getElementUtils().getPackageOf(typeElement).getQualifiedName().toString();

                try {
                    // Crear un nuevo archivo fuente
                    JavaFileObject builderFile = processingEnv.getFiler().createSourceFile(packageName + "." + className + "Generated");
                    try (PrintWriter out = new PrintWriter(builderFile.openWriter())) {
                        // Escribir el código generado
                        out.println("package " + packageName + ";");
                        out.println("public class " + className + "Generated {");
                        out.println("    public String toString() {");
                        out.println("        return \"Generated toString for " + className + "\";");
                        out.println("    }");
                        out.println("}");
                    }
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
        }
        return true;
    }
}

```

#### 3. Uso de la Anotación
Ahora puedes usar la anotación `@GenerateToString` en una clase y observar cómo el procesador genera una nueva clase con el método `toString()`.

```java

@GenerateToString
public class MyClass {
    private String name;
    private int age;

    public MyClass(String name, int age) {
        this.name = name;
        this.age = age;
    }
}

```

#### 4. Resultado

Cuando compiles el proyecto, el procesador generará una nueva clase llamada `MyClassGenerated` en el mismo paquete que `MyClass`. Esta clase contendrá el método `toString()` generado automáticamente.

```java

package com.example;

public class MyClassGenerated {
    public String toString() {
        return "Generated toString for MyClass";
    }
}

```

---

### Nota Importante
El ejemplo anterior genera una **nueva clase** en lugar de modificar la clase original. Esto se debe a que el procesamiento de anotaciones en Java no permite modificar directamente el código fuente de las clases existentes. Sin embargo, puedes lograr un efecto similar utilizando herramientas como Lombok, que interactúan directamente con el compilador para modificar el AST.

Si necesitas modificar la misma clase, podrías considerar el uso de **Bytecode Manipulation Libraries** como ASM o Javassist, pero esto ya no sería parte del procesamiento de anotaciones estándar.

---

### Conclusión
Este ejemplo muestra cómo crear un procesador de anotaciones personalizado que genera código basado en una anotación. Aunque no modifica directamente la clase original, puede generar nuevas clases o archivos relacionados. Para casos más avanzados, puedes explorar herramientas como Lombok o bibliotecas de manipulación de bytecode.

Aquí tienes un ejemplo más avanzado de un procesador de anotaciones en Java que genera **múltiples métodos** dentro de una clase auxiliar basada en la anotación personalizada. Este ejemplo se enfoca en generar varios métodos útiles, como `toString()`, `equals()`, y `hashCode()`, para una clase marcada con una anotación personalizada.

### Ejemplo: Generación de Métodos con Procesamiento de Anotaciones

#### 1. Definición de la Anotación
Define una anotación llamada `@GenerateMethods` que marcará las clases para las cuales queremos generar métodos adicionales.

```java

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Retention(RetentionPolicy.SOURCE) // Solo disponible en tiempo de compilación
@Target(ElementType.TYPE) // Aplicable a clases
public @interface GenerateMethods {
}

```

#### 2. Procesador de Anotaciones
Crea un procesador de anotaciones que genere los métodos `toString()`, `equals()`, y `hashCode()` para la clase anotada.

```java

import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.TypeElement;
import javax.tools.JavaFileObject;
import java.io.IOException;
import java.io.PrintWriter;
import java.util.Set;

@SupportedAnnotationTypes("GenerateMethods")
@SupportedSourceVersion(SourceVersion.RELEASE_8)
public class MethodGeneratorProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (Element element : roundEnv.getElementsAnnotatedWith(GenerateMethods.class)) {
            if (element instanceof TypeElement) {
                TypeElement typeElement = (TypeElement) element;
                String className = typeElement.getSimpleName().toString();
                String packageName = processingEnv.getElementUtils().getPackageOf(typeElement).getQualifiedName().toString();

                try {
                    // Crear un nuevo archivo fuente
                    JavaFileObject builderFile = processingEnv.getFiler().createSourceFile(packageName + "." + className + "Generated");
                    try (PrintWriter out = new PrintWriter(builderFile.openWriter())) {
                        // Escribir el código generado
                        out.println("package " + packageName + ";");
                        out.println("public class " + className + "Generated {");

                        // Método toString()
                        out.println("    public String toString() {");
                        out.println("        return \"Generated toString for " + className + "\";");
                        out.println("    }");

                        // Método equals()
                        out.println("    public boolean equals(Object obj) {");
                        out.println("        if (this == obj) return true;");
                        out.println("        if (obj == null || getClass() != obj.getClass()) return false;");
                        out.println("        return true; // Simplificado para este ejemplo");
                        out.println("    }");

                        // Método hashCode()
                        out.println("    public int hashCode() {");
                        out.println("        return 42; // Valor fijo para este ejemplo");
                        out.println("    }");

                        out.println("}");
                    }
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
        }
        return true;
    }
}

```

#### 3. Uso de la Anotación
Ahora puedes usar la anotación `@GenerateMethods` en una clase y observar cómo el procesador genera una nueva clase con los métodos `toString()`, `equals()`, y `hashCode()`.

```java

@GenerateMethods
public class Person {
    private String name;
    private int age;

    public Person(String name, int age) {
        this.name = name;
        this.age = age;
    }
}

```

#### 4. Resultado
Cuando compiles el proyecto, el procesador generará una nueva clase llamada `PersonGenerated` en el mismo paquete que `Person`. Esta clase contendrá los métodos `toString()`, `equals()`, y `hashCode()` generados automáticamente.

```java

package com.example;

public class PersonGenerated {
    public String toString() {
        return "Generated toString for Person";
    }

    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || getClass() != obj.getClass()) return false;
        return true; // Simplificado para este ejemplo
    }

    public int hashCode() {
        return 42; // Valor fijo para este ejemplo
    }
}

```

### Explicación del Código

1. **Anotación Personalizada
 (`@GenerateMethods`)**:
   - Se utiliza para marcar las clases que necesitan métodos generados.
   - La anotación tiene un alcance de tiempo de compilación (`RetentionPolicy.SOURCE`).

2. **Procesador de Anotaciones (`MethodGeneratorProcessor`)**:
   - Detecta todas las clases anotadas con `@GenerateMethods`.
   - Genera una nueva clase con sufijo `Generated` que contiene los métodos `toString()`, `equals()`, y `hashCode()`.

3. **Generación de Código**:
   - El procesador crea un archivo `.java` dinámicamente durante la compilación.
   - Los métodos generados son simples y pueden ser personalizados según tus necesidades.

4. **Uso de la Anotación**:
   - Al marcar una clase con `@GenerateMethods`, el procesador genera automáticamente una clase auxiliar con los métodos deseados.

---

### Mejoras Posibles

1. **Reflexión sobre Campos**:
   - Puedes mejorar el procesador para analizar los campos de la clase anotada y generar implementaciones más específicas de `toString()`, `equals()`, y `hashCode()`.

2. **Generación en la Misma Clase**:
   - Si deseas modificar la misma clase en lugar de generar una nueva, necesitarás usar herramientas como **Lombok** o manipulación de bytecode (ASM, Javassist).

3. **Métodos Adicionales**:
   - Puedes agregar más métodos, como constructores personalizados, getters/setters, o incluso métodos de validación.

---

### Conclusión

Este ejemplo muestra cómo crear un procesador de anotaciones que genera múltiples métodos en una clase auxiliar. Aunque no modifica directamente la clase original, puede generar nuevas clases con funcionalidades adicionales. Para casos más avanzados, puedes explorar herramientas como Lombok o bibliotecas de manipulación de bytecode.

Un proyecto que utiliza **Java Annotation Processing** para generar código y añadirlo a la misma clase en tiempo de compilación es un caso avanzado, ya que el procesamiento de anotaciones estándar en Java no permite modificar directamente las clases existentes. Sin embargo, se puede lograr un comportamiento similar utilizando herramientas como **Lombok** o mediante el uso de bibliotecas personalizadas que generan código adicional en tiempo de compilación.

A continuación, te muestro un ejemplo básico de cómo podrías implementar un procesador de anotaciones personalizado que genere código y lo "añada" a una clase existente. Este ejemplo no modifica directamente la clase original, pero genera código complementario que se integra con la clase existente.

### Ejemplo: Procesador de Anotaciones Personalizado

#### 1. Definición de la Anotación
Primero, creamos una anotación personalizada llamada `@AddLogging`.

```java

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Retention(RetentionPolicy.SOURCE) // Solo disponible en tiempo de compilación
@Target(ElementType.METHOD)       // Aplicable solo a métodos
public @interface AddLogging {
}

```

Esta anotación se usará para marcar métodos en los que queremos añadir automáticamente registros (logs).

---

#### 2. Procesador de Anotaciones
A continuación, creamos un procesador de anotaciones que detecte los métodos anotados con `@AddLogging` y genere código para añadir registros antes y después de la ejecución del método.

```java

import javax.annotation.processing.*;
import javax.lang.model.element.*;
import javax.tools.Diagnostic;
import javax.tools.JavaFileObject;
import java.io.IOException;
import java.io.PrintWriter;
import java.util.Set;

@SupportedAnnotationTypes("AddLogging")
@SupportedSourceVersion(SourceVersion.RELEASE_8)
public class LoggingProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (Element element : roundEnv.getElementsAnnotatedWith(AddLogging.class)) {
            if (element.getKind() == ElementKind.METHOD) {
                ExecutableElement method = (ExecutableElement) element;
                String className = ((TypeElement) method.getEnclosingElement()).getQualifiedName().toString();
                String methodName = method.getSimpleName().toString();

                try {
                    generateLoggingCode(className, methodName);
                } catch (IOException e) {
                    processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, "Error generating code: " + e.getMessage());
                }
            }
        }
        return true;
    }

    private void generateLoggingCode(String className, String methodName) throws IOException {
        String newClassName = className + "Logger";
        JavaFileObject file = processingEnv.getFiler().createSourceFile(newClassName);

        try (PrintWriter out = new PrintWriter(file.openWriter())) {
            out.println("public class " + newClassName + " {");
            out.println("    public void " + methodName + "() {");
            out.println("        System.out.println(\"Entering method: " + methodName + "\");");
            out.println("        // Original method logic would go here");
            out.println("        System.out.println(\"Exiting method: " + methodName + "\");");
            out.println("    }");
            out.println("}");
        }
    }
}

```

Este procesador:
1. Detecta métodos anotados con `@AddLogging`.
2. Genera una nueva clase con el sufijo `Logger` que contiene una versión modificada del método con registros.

```java
    }

```

Proyecto de procesamiento de anotaciones Java que genera código en tiempo de compilación
Este proyecto demuestra cómo usar el procesamiento de anotaciones de Java para generar código y agregarlo a la misma clase en tiempo de compilación.
Dependencias
Asegúrate de tener las siguientes dependencias en tu proyecto:
 * Java Compiler API
 * Un procesador de anotaciones (puedes crear el tuyo propio o usar uno existente como Auto)
Anotación
Crea una anotación personalizada que marcará los elementos para los que se debe generar código. Por ejemplo:
import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Retention(RetentionPolicy.SOURCE)
@Target(ElementType.TYPE)
public @interface GenerarCodigo {
    String valor();
}

Procesador de anotaciones
Crea un procesador de anotaciones que se encargará de generar el código. Este procesador buscará la anotación @GenerarCodigo y agregará el código correspondiente a la clase.
import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.TypeElement;
import javax.tools.JavaFileObject;
import javax.tools.StandardLocation;
import java.io.IOException;
import java.io.Writer;
import java.util.Set;

@SupportedAnnotationTypes("GenerarCodigo")
@SupportedSourceVersion(javax.lang.model.SourceVersion.RELEASE_17)
public class CodigoProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (Element annotatedElement : roundEnv.getElementsAnnotatedWith(GenerarCodigo.class)) {
            TypeElement typeElement = (TypeElement) annotatedElement;
            String valor = typeElement.getAnnotation(GenerarCodigo.class).valor();

            try {
                JavaFileObject sourceFile = processingEnv.getFiler().createSourceFile(StandardLocation.SOURCE_OUTPUT, typeElement.getQualifiedName());
                Writer writer = sourceFile.openWriter();

                writer.write("package " + typeElement.getPackageName() + ";\n");
                writer.write("\n");
                writer.write("public class " + typeElement.getSimpleName() + " {\n");
                writer.write("\n");
                writer.write("    public String obtenerValor() {\n");
                writer.write("        return \"" + valor + "\";\n");
                writer.write("    }\n");
                writer.write("}\n");

                writer.close();
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return true;
    }
}

Uso
Para usar la anotación, simplemente agrega @GenerarCodigo a la clase que deseas modificar. Por ejemplo:
@GenerarCodigo(valor = "Este es el valor generado")
public class MiClase {
    // ...
}

Al compilar el proyecto, el procesador de anotaciones generará el código y lo agregará a la clase MiClase.

Crea un procesador de anotaciones llamado ConstructorProcessor que se encargará de generar el código del constructor:
import com.google.auto.service.AutoService;
import com.squareup.javapoet.ClassName;
import com.squareup.javapoet.JavaFile;
import com.squareup.javapoet.MethodSpec;
import com.squareup.javapoet.TypeSpec;

import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.Processor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.TypeElement;
import javax.tools.JavaFileObject;
import javax.tools.StandardLocation;
import java.io.IOException;
import java.io.Writer;
import java.util.Set;

@AutoService(Processor.class)
@SupportedAnnotationTypes("GenerarConstructor")
@SupportedSourceVersion(javax.lang.model.SourceVersion.RELEASE_17)
public class ConstructorProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (Element annotatedElement : roundEnv.getElementsAnnotatedWith(GenerarConstructor.class)) {
            TypeElement typeElement = (TypeElement) annotatedElement;
            String className = typeElement.getSimpleName().toString();
            String packageName = processingEnv.getElementUtils().getPackageOf(typeElement).toString();

            MethodSpec constructor = MethodSpec.constructorBuilder()
                    .addModifiers(javax.lang.model.element.Modifier.PUBLIC)
                    .addStatement("// Constructor generado automáticamente")
                    .build();

            TypeSpec classBuilder = TypeSpec.classBuilder(className)
                    .addModifiers(javax.lang.model.element.Modifier.PUBLIC)
                    .addMethod(constructor)
                    .build();

            JavaFile javaFile = JavaFile.builder(packageName, classBuilder)
                    .build();

            try {
                JavaFileObject sourceFile = processingEnv.getFiler().createSourceFile(StandardLocation.SOURCE_OUTPUT, typeElement.getQualifiedName());
                Writer writer = sourceFile.openWriter();
                javaFile.writeTo(writer);
                writer.close();
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return true;
    }
}

Al compilar el proyecto, el procesador de anotaciones generará el constructor y lo agregará a la clase MiClase.

En Java, el **Annotation Processing** está diseñado para generar código adicional en tiempo de compilación, pero no permite modificar directamente las clases existentes. Esto se debe a que el procesador de anotaciones trabaja con un modelo inmutable del código fuente y no tiene acceso directo al AST (Abstract Syntax Tree) de las clases existentes.

Sin embargo, hay formas de lograr un comportamiento similar mediante el uso de herramientas como **Lombok**, que utiliza técnicas avanzadas (como la manipulación del bytecode) para "modificar" las clases existentes. Si deseas implementar algo similar sin depender de Lombok, puedes usar una combinación de **Annotation Processing** y **Bytecode Manipulation** (por ejemplo, con **Javassist** o **ASM**).

A continuación, te muestro un proyecto que utiliza **Annotation Processing** junto con **Bytecode Manipulation** para generar código y añadirlo a la misma clase en tiempo de compilación.

---

### Proyecto: Generación de Código en la Misma Clase

#### 1. Definición de la Anotación
Primero, creamos una anotación personalizada llamada `@AddLogging`.

```java
import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Retention(RetentionPolicy.CLASS) // Disponible en tiempo de compilación y runtime
@Target(ElementType.METHOD)       // Aplicable solo a métodos
public @interface AddLogging {
}
```

Esta anotación se usará para marcar métodos en los que queremos añadir automáticamente registros (logs).

---

#### 2. Procesador de Anotaciones
El procesador de anotaciones detectará los métodos anotados con `@AddLogging` y generará un archivo auxiliar que indica qué métodos deben ser modificados. Este archivo será utilizado posteriormente por una herramienta de manipulación de bytecode.

```java
import javax.annotation.processing.*;
import javax.lang.model.element.*;
import javax.tools.Diagnostic;
import java.io.IOException;
import java.io.Writer;
import java.util.Set;

@SupportedAnnotationTypes("AddLogging")
@SupportedSourceVersion(SourceVersion.RELEASE_8)
public class LoggingProcessor extends AbstractProcessor {

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (Element element : roundEnv.getElementsAnnotatedWith(AddLogging.class)) {
            if (element.getKind() == ElementKind.METHOD) {
                ExecutableElement method = (ExecutableElement) element;
                String className = ((TypeElement) method.getEnclosingElement()).getQualifiedName().toString();
                String methodName = method.getSimpleName().toString();

                try {
                    writeLoggingMetadata(className, methodName);
                } catch (IOException e) {
                    processingEnv.getMessager().printMessage(Diagnostic.Kind.ERROR, "Error writing metadata: " + e.getMessage());
                }
            }
        }
        return true;
    }

    private void writeLoggingMetadata(String className, String methodName) throws IOException {
        String metadataFile = className.replace('.', '/') + ".logging";
        Writer writer = processingEnv.getFiler().createResource(StandardLocation.CLASS_OUTPUT, "", metadataFile).openWriter();
        writer.write(methodName);
        writer.close();
    }
}
```

Este procesador:
1. Detecta métodos anotados con `@AddLogging`.
2. Escribe un archivo de metadatos que contiene el nombre del método a modificar.

---

#### 3. Manipulación de Bytecode con Javassist
Usaremos **Javassist** para modificar el bytecode de las clases existentes después de que el compilador haya generado los archivos `.class`. Esto se hace en un paso posterior al procesamiento de anotaciones.

Agrega la dependencia de Javassist en tu `pom.xml` si usas Maven:

```xml
<dependency>
    <groupId>org.javassist</groupId>
    <artifactId>javassist</artifactId>
    <version>3.29.2-GA</version>
</dependency>
```

Ahora, crea una herramienta que lea los metadatos generados por el procesador de anotaciones y modifique las clases correspondientes:

```java
import javassist.*;

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class BytecodeModifier {

    public static void main(String[] args) throws Exception {
        String className = "ExampleClass"; // Nombre de la clase a modificar
        ClassPool pool = ClassPool.getDefault();
        CtClass ctClass = pool.get(className);

        // Leer el archivo de metadatos generado por el procesador
        String metadataFile = className.replace('.', '/') + ".logging";
        try (BufferedReader reader = new BufferedReader(new FileReader(metadataFile))) {
            String methodName = reader.readLine();

            // Modificar el método añadiendo registros
            CtMethod method = ctClass.getDeclaredMethod(methodName);
            method.insertBefore("System.out.println(\"Entering method: " + methodName + "\");");
            method.insertAfter("System.out.println(\"Exiting method: " + methodName + "\");");
        }

        // Guardar la clase modificada
        ctClass.writeFile();
        System.out.println("Class modified successfully!");
    }
}
```

Este programa:
1. Lee el archivo de metadatos generado por el procesador de anotaciones.
2. Usa Javassist para modificar el bytecode de la clase existente, añadiendo registros antes y después del método.
