
Para añadir endpoints `GET` que permitan obtener los atributos individuales de una clase `Persona`, podemos crear endpoints específicos para cada atributo. Por ejemplo, podríamos tener:

1. **GET /personas/{id}/nombre** - Obtener el nombre de una persona por su ID.
2. **GET /personas/{id}/edad** - Obtener la edad de una persona por su ID.

Estos endpoints devolverán únicamente el valor del atributo solicitado en lugar de toda la información de la persona.

### Implementación Completa con Endpoints Adicionales

A continuación, se muestra cómo extender el servidor HTTP anterior para incluir estos nuevos endpoints.

```java
import com.sun.net.httpserver.HttpServer;
import com.sun.net.httpserver.HttpHandler;
import com.sun.net.httpserver.HttpExchange;

import java.io.IOException;
import java.io.OutputStream;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class PersonaHttpServer {

    // Almacenamiento en memoria de las personas
    private static Map<Integer, Persona> personasDB = new ConcurrentHashMap<>();

    public static void main(String[] args) throws IOException {
        HttpServer server = HttpServer.create(new InetSocketAddress(8080), 0);

        // Definir los contextos para cada endpoint
        server.createContext("/personas", new GetAllPersonasHandler());
        server.createContext("/personas/", new PersonaHandler()); // Para GET, PUT, DELETE con ID
        server.createContext("/personas", new CreatePersonaHandler()); // Para POST

        // Nuevos endpoints para atributos individuales
        server.createContext("/personas/", new GetNombreHandler()); // GET /personas/{id}/nombre
        server.createContext("/personas/", new GetEdadHandler());   // GET /personas/{id}/edad

        server.setExecutor(null); // Usa el executor por defecto
        server.start();
        System.out.println("Servidor iniciado en el puerto 8080");
    }

    // Handler para obtener todas las personas
    static class GetAllPersonasHandler implements HttpHandler {
        @Override
        public void handle(HttpExchange exchange) throws IOException {
            if ("GET".equalsIgnoreCase(exchange.getRequestMethod())) {
                StringBuilder response = new StringBuilder();
                for (Persona persona : personasDB.values()) {
                    response.append(persona.toString()).append("\n");
                }

                exchange.sendResponseHeaders(200, response.toString().getBytes().length);
                OutputStream os = exchange.getResponseBody();
                os.write(response.toString().getBytes());
                os.close();
            } else {
                exchange.sendResponseHeaders(405, -1); // Método no permitido
            }
        }
    }

    // Handler para manejar GET, PUT, DELETE por ID
    static class PersonaHandler implements HttpHandler {
        @Override
        public void handle(HttpExchange exchange) throws IOException {
            String path = exchange.getRequestURI().getPath();
            String[] pathParts = path.split("/");
            int id = Integer.parseInt(pathParts[pathParts.length - 1]);

            switch (exchange.getRequestMethod()) {
                case "GET":
                    handleGetPersona(exchange, id);
                    break;
                case "PUT":
                    handleUpdatePersona(exchange, id);
                    break;
                case "DELETE":
                    handleDeletePersona(exchange, id);
                    break;
                default:
                    exchange.sendResponseHeaders(405, -1); // Método no permitido
            }
        }

        private void handleGetPersona(HttpExchange exchange, int id) throws IOException {
            Persona persona = personasDB.get(id);
            if (persona != null) {
                String response = persona.toString();
                exchange.sendResponseHeaders(200, response.getBytes().length);
                OutputStream os = exchange.getResponseBody();
                os.write(response.getBytes());
                os.close();
            } else {
                exchange.sendResponseHeaders(404, -1); // No encontrado
            }
        }

        private void handleUpdatePersona(HttpExchange exchange, int id) throws IOException {
            Persona persona = personasDB.get(id);
            if (persona != null) {
                String requestBody = new String(exchange.getRequestBody().readAllBytes());
                Map<String, String> params = queryToMap(requestBody);
                persona.setNombre(params.getOrDefault("nombre", persona.getNombre()));
                persona.setEdad(Integer.parseInt(params.getOrDefault("edad", String.valueOf(persona.getEdad()))));

                exchange.sendResponseHeaders(200, -1); // Actualizado correctamente
            } else {
                exchange.sendResponseHeaders(404, -1); // No encontrado
            }
        }

        private void handleDeletePersona(HttpExchange exchange, int id) throws IOException {
            if (personasDB.remove(id) != null) {
                exchange.sendResponseHeaders(200, -1); // Eliminado correctamente
            } else {
                exchange.sendResponseHeaders(404, -1); // No encontrado
            }
        }

        private Map<String, String> queryToMap(String query) {
            Map<String, String> result = new HashMap<>();
            if (query != null) {
                for (String param : query.split("&")) {
                    String[] entry = param.split("=");
                    if (entry.length > 1) {
                        result.put(entry[0], entry[1]);
                    } else {
                        result.put(entry[0], "");
                    }
                }
            }
            return result;
        }
    }

    // Handler para crear una nueva persona
    static class CreatePersonaHandler implements HttpHandler {
        @Override
        public void handle(HttpExchange exchange) throws IOException {
            if ("POST".equalsIgnoreCase(exchange.getRequestMethod())) {
                String requestBody = new String(exchange.getRequestBody().readAllBytes());
                Map<String, String> params = queryToMap(requestBody);
                String nombre = params.get("nombre");
                int edad = Integer.parseInt(params.get("edad"));

                // Generar un nuevo ID
                int id = personasDB.size() + 1;
                Persona nuevaPersona = new Persona(id, nombre, edad);
                personasDB.put(id, nuevaPersona);

                String response = "Persona creada: " + nuevaPersona.toString();
                exchange.sendResponseHeaders(201, response.getBytes().length);
                OutputStream os = exchange.getResponseBody();
                os.write(response.getBytes());
                os.close();
            } else {
                exchange.sendResponseHeaders(405, -1); // Método no permitido
            }
        }

        private Map<String, String> queryToMap(String query) {
            Map<String, String> result = new HashMap<>();
            if (query != null) {
                for (String param : query.split("&")) {
                    String[] entry = param.split("=");
                    if (entry.length > 1) {
                        result.put(entry[0], entry[1]);
                    } else {
                        result.put(entry[0], "");
                    }
                }
            }
            return result;
        }
    }

    // Handler para obtener el nombre de una persona por ID
    static class GetNombreHandler implements HttpHandler {
        @Override
        public void handle(HttpExchange exchange) throws IOException {
            if ("GET".equalsIgnoreCase(exchange.getRequestMethod())) {
                String path = exchange.getRequestURI().getPath();
                String[] pathParts = path.split("/");
                int id = Integer.parseInt(pathParts[pathParts.length - 2]); // El ID está antes de "/nombre"

                Persona persona = personasDB.get(id);
                if (persona != null) {
                    String response = persona.getNombre();
                    exchange.sendResponseHeaders(200, response.getBytes().length);
                    OutputStream os = exchange.getResponseBody();
                    os.write(response.getBytes());
                    os.close();
                } else {
                    exchange.sendResponseHeaders(404, -1); // No encontrado
                }
            } else {
                exchange.sendResponseHeaders(405, -1); // Método no permitido
            }
        }
    }

    // Handler para obtener la edad de una persona por ID
    static class GetEdadHandler implements HttpHandler {
        @Override
        public void handle(HttpExchange exchange) throws IOException {
            if ("GET".equalsIgnoreCase(exchange.getRequestMethod())) {
                String path = exchange.getRequestURI().getPath();
                String[] pathParts = path.split("/");
                int id = Integer.parseInt(pathParts[pathParts.length - 2]); // El ID está antes de "/edad"

                Persona persona = personasDB.get(id);
                if (persona != null) {
                    String response = String.valueOf(persona.getEdad());
                    exchange.sendResponseHeaders(200, response.getBytes().length);
                    OutputStream os = exchange.getResponseBody();
                    os.write(response.getBytes());
                    os.close();
                } else {
                    exchange.sendResponseHeaders(404, -1); // No encontrado
                }
            } else {
                exchange.sendResponseHeaders(405, -1); // Método no permitido
            }
        }
    }
}
```

### Explicación de los Nuevos Endpoints

1. **GET /personas/{id}/nombre**: Este endpoint devuelve únicamente el nombre de la persona con el ID especificado.
2. **GET /personas/{id}/edad**: Este endpoint devuelve únicamente la edad de la persona con el ID especificado.

### Ejemplo de Uso

- **Crear una nueva persona**:
  ```bash
  curl -X POST -d "nombre=Juan&edad=30" http://localhost:8080/personas
  ```

- **Obtener el nombre de una persona por ID**:
  ```bash
  curl http://localhost:8080/personas/1/nombre
  ```

- **Obtener la edad de una persona por ID**:
  ```bash
  curl http://localhost:8080/personas/1/edad
  ```

### Resumen

En este ejemplo, hemos extendido el servidor HTTP básico para incluir endpoints adicionales que permiten obtener los atributos individuales de una persona (`nombre` y `edad`). Estos endpoints son útiles cuando solo necesitas acceder a un atributo específico de una entidad en lugar de toda la información.

Este enfoque es modular y permite que el servidor sea más flexible, ya que puedes agregar más endpoints para otros atributos o funcionalidades según sea necesario.
